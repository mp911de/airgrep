# Requirements cannot depend on sdkconfig, the sources can: the benchmarks of the conflict engine
# exist where the engine does, in a build with a display.
set(srcs "hwtest.cpp")
if (CONFIG_AIRGREP_DISPLAY)
    list(APPEND srcs "scenario.cpp" "benchmark.cpp")
endif ()
if (CONFIG_AIRGREP_BOARD_PLUS AND CONFIG_AIRGREP_DISPLAY STREQUAL "800x480-touch")
    list(APPEND srcs "render_benchmark.cpp" "render_profile.cpp")
    set(render_benchmark ON)
endif ()
idf_component_register(SRCS ${srcs} INCLUDE_DIRS "include" PRIV_REQUIRES loop airgrep esp_timer esp_system esp_app_format display daidalus)
if (CONFIG_AIRGREP_DISPLAY)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE HWTEST_BENCHMARKS=1)
endif ()
if (render_benchmark)
    idf_component_get_property(display_dir display COMPONENT_DIR)
    target_include_directories(${COMPONENT_LIB} PRIVATE "${display_dir}/ui/widgets/radar")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE HWTEST_RENDER=1)
    target_link_options(${COMPONENT_LIB} INTERFACE "LINKER:--wrap=radar_outline_text")
endif ()
